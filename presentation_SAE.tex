\begin{frame}{Detection de visage --- MediaPipe}

\begin{columns}[T]
\begin{column}{0.55\textwidth}
\begin{block}{Pourquoi c'est necessaire ?}
Nos modeles ont ete entraines sur des \textbf{visages recadres} (dataset UTKFace). En production, l'utilisateur prend une photo du \textbf{corps entier} ou dans un contexte reel.

\vspace{0.2cm}
MediaPipe \textbf{detecte et isole le visage} automatiquement avant de l'envoyer au modele.
\end{block}
\end{column}

\begin{column}{0.4\textwidth}
\begin{exampleblock}{Comment ca marche}
\begin{itemize}
    \item Detection en 15-30ms
    \item Fonctionne meme de loin
    \item Recadrage carre du visage
    \item Guide ovale en temps reel
    \item Rectangle vert quand un visage est detecte
\end{itemize}
\end{exampleblock}
\end{column}
\end{columns}

\vspace{0.2cm}
\centering
{\small\color{accent} Sans cette etape, le modele analyserait les vetements ou l'arriere-plan au lieu du visage}
\end{frame}

% =============================================
\begin{frame}{Comment le modele predit-il ?}

\begin{exampleblock}{Le principe}
Le modele a \textbf{appris par l'exemple} a partir de 23 708 visages. Il detecte des \textbf{patterns} dans les pixels, comme un enfant qui apprend a reconnaitre les visages.
\end{exampleblock}

\vspace{0.2cm}

\begin{columns}[T]
\begin{column}{0.3\textwidth}
\begin{block}{\centering Age}
\centering
{\small Rides, texture de peau, cheveux gris, rondeur du visage}
\end{block}
\end{column}
\begin{column}{0.3\textwidth}
\begin{block}{\centering Genre}
\centering
{\small Machoire, pilosite faciale, structure osseuse}
\end{block}
\end{column}
\begin{column}{0.3\textwidth}
\begin{block}{\centering Ethnicite}
\centering
{\small Teint, forme des yeux et du nez, proportions}
\end{block}
\end{column}
\end{columns}

\vspace{0.2cm}
\centering
{\small\color{accent} Pas d'infrarouge ni de capteur special --- uniquement les pixels}

\begin{frame}{Le dataset UTKFace}

\begin{columns}[T]
\begin{column}{0.55\textwidth}

\begin{itemize}
    \item \textbf{Age} : 0 a 116 ans
    \item \textbf{Genre} : Homme / Femme

\begin{exampleblock}{Preprocessing}
{\small Resize 224x224 --- Augmentation (flip, crop, luminosite) --- Mixed precision float16}

\begin{frame}{Strategie 1 --- Trois modeles specialises}

\begin{exampleblock}{Principe}
{\small Un modele dedie par tache avec un backbone \textbf{EfficientNetB0} pre-entraine sur ImageNet.}
\end{exampleblock}

\vspace{0.2cm}

\begin{columns}[T]
\begin{column}{0.3\textwidth}
\begin{block}{\centering Age}
\centering
{\small Sortie lineaire\\Huber Loss\\[0.1cm]\textbf{MAE $\approx$ 6.2 ans}}
\end{block}
\end{column}
\begin{column}{0.3\textwidth}
\begin{block}{\centering Genre}
\centering
{\small Sortie sigmoid\\Binary Cross-Entropy\\[0.1cm]\textbf{Acc $\approx$ 90\%}}
\end{block}
\end{column}
\begin{column}{0.3\textwidth}
\begin{block}{\centering Ethnicite}
\centering
{\small Sortie softmax (5)\\Focal Loss\\[0.1cm]\textbf{Acc $\approx$ 70\%}}
\end{block}
\end{column}
\end{columns}

\vspace{0.2cm}
\centering
{\small\color{steel} Entrainement : warmup (backbone gele) puis fine-tuning (15 couches)}
\end{frame}

% =============================================
\begin{frame}{Strategie 2 --- Modele multi-tache}

\begin{block}{Principe}
{\small Un seul modele EfficientNetB0 predit les 3 taches simultanement. Le backbone est partage, seules les couches de sortie (Dense) sont specifiques a chaque tache.}
\end{block}

\vspace{0.3cm}

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{exampleblock}{Avantage}
{\small Un seul fichier, inference plus rapide, le modele apprend des \textbf{features communes} aux 3 taches.}
\end{exampleblock}
\end{column}
\begin{column}{0.48\textwidth}
\begin{alertblock}{Inconvenient}
{\small Les taches se \textbf{partagent la capacite} du modele. Chaque tache individuelle est legerement moins bonne qu'un modele specialise.}
\end{alertblock}
\end{column}
\end{columns}
\end{frame}

% =============================================
\begin{frame}{Mixture of Experts --- Initiative personnelle}

\begin{exampleblock}{Pourquoi le MoE ?}
{\small Apres nos recherches, nous avons voulu aller \textbf{au-dela du sujet} en testant une architecture Mixture of Experts pour ameliorer le modele multi-tache.}
\end{exampleblock}

\vspace{0.2cm}

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{block}{Comment ca marche}
\begin{itemize}
    \item {\small 1 backbone EfficientNetB0 partage}
    \item {\small 3 tetes avec \textbf{4 experts} chacune}
    \item {\small 1 \textbf{gating network} par tache}
    \item {\small Le gating choisit l'expert le plus adapte pour chaque visage}
\end{itemize}
\end{block}
\end{column}
\begin{column}{0.48\textwidth}
\begin{block}{Resultat}
\begin{itemize}
    \item {\small Genre : \textbf{90.7\%} accuracy}
    \item {\small Ethnicite : \textbf{70.8\%} accuracy}
    \item {\small Age : \textbf{6.36 ans} MAE}
    \item {\small Meilleur que le multi-tache simple}
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Strategie 3 --- Comparaison des architectures}

\begin{center}
{\small
\begin{tabular}{lccccc}
\toprule
\textbf{Architecture} & \textbf{Genre} & \textbf{Ethnicite} & \textbf{Age MAE} & \textbf{Taille} & \textbf{Score} \\
\midrule
\textbf{EfficientNetB0} & \textbf{90.1\%} & \textbf{68.3\%} & 6.19 & 5M & \textbf{9.17} \\
ResNet50 & 89.9\% & 66.9\% & \textbf{5.87} & 30M & 9.15 \\
MobileNetV3 & 88.5\% & 67.8\% & 7.11 & 4M & 9.02 \\
MobileNetV2 & 85.0\% & 55.7\% & 7.59 & 3M & 8.37 \\
\bottomrule
\end{tabular}
}
\end{center}

\vspace{0.3cm}

\begin{exampleblock}{Pourquoi EfficientNetB0 ?}
{\small Meilleur compromis \textbf{precision / taille}. ResNet50 est quasi egal mais 6x plus gros et 2x plus lent.}
\end{exampleblock}
\end{frame}

% =============================================
\begin{frame}{Avantages et inconvenients de chaque mode}

{\small
\begin{center}
\begin{tabular}{p{2.5cm}p{4.5cm}p{4.5cm}}
\toprule
\textbf{Mode} & \textbf{Avantages} & \textbf{Inconvenients} \\
\midrule
\textbf{Hybride} (defaut) & Meilleure precision grace au TTA (flip + moyenne) & Charge 3 modeles en memoire, plus lent \\
\midrule
\textbf{3 Specialises} & Chaque modele optimise pour sa tache & 3 fichiers a charger, pas de partage de features \\
\midrule
\textbf{MoE} & 1 seul fichier, experts specialises par type de visage & Necessite une librairie supplementaire (select-tf-ops) \\
\bottomrule
\end{tabular}
\end{center}
}

\vspace{0.2cm}
\centering
{\small\color{steel} Le mode \textbf{Hybride} est recommande : il combine les 3 modeles specialises avec du Test Time Augmentation pour des resultats plus stables.}
\end{frame}

\begin{frame}{Processus d'entrainement}

\begin{block}{Pipeline}
{\small
\begin{enumerate}
\setlength{\itemsep}{4pt}
    \item \textbf{Comparaison des backbones} --- On teste 4 architectures (MobileNetV2, EfficientNetB0, ResNet50, MobileNetV3) sur les 3 taches. EfficientNetB0 gagne.
    \item \textbf{Entrainement multi-tache MoE} --- On entraine le modele Mixture of Experts avec EfficientNetB0 en 3 phases (warmup, fine-tune partiel, fine-tune etendu).
    \item \textbf{Entrainement des 3 specialises} --- Chaque modele (age, genre, ethnicite) est entraine separement avec le meme backbone.
    \item \textbf{Conversion TFLite} --- Les modeles Keras sont convertis en TensorFlow Lite (float16) pour le deploiement mobile.
    \item \textbf{Integration Android} --- Les fichiers .tflite sont embarques dans l'application.
\end{enumerate}
}
\end{block}

\vspace{0.2cm}
\centering
{\small\color{steel} Entrainement realise en local sur GPU NVIDIA RTX 4050 (6 Go VRAM) --- environ 3h au total}
\end{frame}

\begin{frame}{Difficultes rencontrees}

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\begin{alertblock}{GPU limite (6 Go VRAM)}
{\small Crashs frequents. Batch reduit a 4, couches debloquees par paliers, nettoyage memoire.}
\end{alertblock}

\vspace{0.1cm}

\begin{alertblock}{Reproductibilite}
{\small Resultats variables. Seeds fixes pour numpy, tensorflow, random.}
\end{alertblock}
\end{column}

\begin{column}{0.48\textwidth}
\begin{alertblock}{Dataset desequilibre}
{\small Classe ``Autre'' = 5\%. Focal Loss + poids de classe.}
\end{alertblock}

\vspace{0.1cm}

\begin{alertblock}{Confusion Asian / Indian}
{\small Traits proches. Limite du dataset, ameliorable avec plus de donnees.}
\end{alertblock}
\end{column}
\end{columns}
